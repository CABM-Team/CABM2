from PIL import Image

# 打开两个图片
img2 = Image.open('2.png').convert('RGBA')  # 确保有 Alpha 通道
img3 = Image.open('3.png').convert('RGBA')  # 部分透明图片

# 获取 3.png 的 Alpha 通道（透明信息）
alpha3 = img3.split()[-1]  # 最后一个通道是 Alpha

# 将 3.png 的 Alpha 通道应用到 2.png 上
img2.putalpha(alpha3)

# 保存结果
img2.save('output.png')

print("已将 3.png 的透明区域应用到 2.png，保存为 output.png")
